import re, json

def extract_gaia_content(result):
    final_contents = []

    for raw in result:  # raw is a str like above
        # Extract the text=... part with regex
        match = re.search(r'Part\(text="?(.+?)"?\)', raw)
        if not match:
            continue

        txt = match.group(1).strip()

        # Normalize quotes for JSON
        safe_txt = txt.replace("'", '"')

        # Case 1: Fetcher
        if "status" in safe_txt and "data" in safe_txt and "result" in safe_txt:
            try:
                parsed = json.loads(safe_txt)
                final_contents.append({
                    "status": parsed.get("status"),
                    "results": parsed.get("data", {}).get("result")
                })
            except Exception:
                final_contents.append(safe_txt)

        # Case 2: Summarizer
        elif "topic" in safe_txt and "Summary_sentence_1" in safe_txt:
            try:
                parsed = json.loads(safe_txt)
                summary_only = {"topic": parsed.get("topic")}
                for k, v in parsed.items():
                    if k.lower().startswith("summary_sentence"):
                        summary_only[k] = v
                final_contents.append(summary_only)
            except Exception:
                final_contents.append(safe_txt)

    return final_contents
