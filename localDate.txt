<!-- Display ISO 8601 formatted date only if not null -->
<input type="hidden"
       name="startDatetime"
       th:if="${startDatetime != null}"
       th:value="${#temporals.format(startDatetime, 'yyyy-MM-dd''T''HH:mm')}" />

<!-- Fallback to empty string if null -->
<input type="hidden"
       name="startDatetime"
       th:unless="${startDatetime != null}"
       value="" />
