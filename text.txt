import os
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

# ✅ Optimize CPU Performance
os.environ["OMP_NUM_THREADS"] = "2"  # Limit CPU threads

# 🔹 Update with your actual model directory
MODEL_PATH = "C:/models/phi3-mini-128k/"

# ✅ Load tokenizer
tokenizer = AutoTokenizer.from_pretrained(MODEL_PATH)

# ✅ Load model (Only Once)
model = AutoModelForCausalLM.from_pretrained(
    MODEL_PATH,
    torch_dtype=torch.float32,  # CPU-friendly format
    device_map="auto",
    trust_remote_code=True
)

# ✅ Function to Process Sentences
def process_text(input_text):
    formatted_input = f'Fix the grammar and return ONLY the corrected sentence in the same format:\n"{input_text}"\nCorrected:'

    input_ids = tokenizer(formatted_input, return_tensors="pt").input_ids.to(model.device)

    # ✅ Generate corrected text with strict stopping conditions
    output_ids = model.generate(
        input_ids,
        max_new_tokens=50,  # Keep it short to prevent unnecessary text
        do_sample=False,  
        num_beams=1,      
        repetition_penalty=2.0,  # EVEN STRONGER penalty to prevent repeated phrases  
        eos_token_id=tokenizer.eos_token_id,  
        return_dict_in_generate=False  
    )
    
    revised_text = tokenizer.decode(output_ids[0], skip_special_tokens=True)

    # ✅ Ensure spacing is fixed
    revised_text = revised_text.replace("\n", " ").strip()
    
    return revised_text

# 🔹 Keep running multiple test cases
while True:
    input_text = input("\nEnter a sentence to correct (or type 'exit' to stop): ")
    if input_text.lower() == "exit":
        break

    revised_text = process_text(input_text)
    
    print("\n🔹 Original Text:", input_text)
    print("✅ Revised Text:", revised_text)
