 @MockBean
    private Clock clock;

    private MockHttpSession getSession(String location) {
        MockHttpSession session = new MockHttpSession();
        session.setAttribute("username", "testuser");
        session.setAttribute("team", "DMC");
        session.setAttribute("location", location);
        session.setAttribute("role", "ADMIN");
        return session;
    }

    private void setMockTime(int hour, int minute) {
        LocalDateTime mockTime = LocalDateTime.of(2025, 4, 10, hour, minute);
        Instant instant = mockTime.atZone(ZoneId.of("America/New_York")).toInstant();
        when(clock.instant()).thenReturn(instant);
        when(clock.getZone()).thenReturn(ZoneId.of("America/New_York"));
    }

    @Test
    void testGetHandover_USUser_MorningWindow() throws Exception {
        setMockTime(6, 0); // 6:00 AM ET

        when(handoverService.getHandoverByShift(anyString(), any(), any())).thenReturn(Optional.empty());
        when(handoverService.getHandoverHeading("US")).thenReturn("US Shift");

        mockMvc.perform(get("/api/handover")
                .session(getSession("US")))
                .andExpect(status().isOk());
    }
