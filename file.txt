import torch
import numpy as np
import onnx
import onnxruntime
from transformers import AutoTokenizer, AutoModel

# ✅ Define paths
MODEL_DIR = "c:/users/abc/downloads/code-baton/jina-embedding-s-en-v1"  # Update to your model path
ONNX_MODEL_PATH = "c:/users/abc/downloads/code-baton/sentence_transformer.onnx"

# ✅ Load tokenizer and model
print(f"🔄 Loading model from {MODEL_DIR}...")
tokenizer = AutoTokenizer.from_pretrained(MODEL_DIR)
model = AutoModel.from_pretrained(MODEL_DIR)

# ✅ Ensure model is in eval mode
model.eval()

# ✅ Create dummy input sentence & tokenize it
dummy_text = ["This is a test sentence."]
inputs = tokenizer(dummy_text, return_tensors="pt")

# ✅ Extract input tensors
input_ids = inputs["input_ids"]
attention_mask = inputs["attention_mask"]

# ✅ Export model to ONNX
torch.onnx.export(
    model,
    (input_ids, attention_mask),  # Pass correct inputs
    ONNX_MODEL_PATH,
    export_params=True,
    opset_version=13,
    do_constant_folding=True,
    input_names=["input_ids", "attention_mask"],
    output_names=["last_hidden_state"],
    dynamic_axes={
        "input_ids": {0: "batch_size", 1: "sequence_length"},
        "attention_mask": {0: "batch_size", 1: "sequence_length"},
        "last_hidden_state": {0: "batch_size", 1: "sequence_length"},
    },
)

print(f"✅ ONNX conversion completed successfully! Model saved at {ONNX_MODEL_PATH}")
