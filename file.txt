import torch
import onnx
import numpy as np
import onnxruntime
from sentence_transformers import SentenceTransformer

# ✅ Define paths
MODEL_DIR = "c:/users/abc/downloads/code-baton/jina-embedding-s-en-v1"  # Your model path
ONNX_MODEL_PATH = "c:/users/abc/downloads/code-baton/sentence_transformer.onnx"

# ✅ Load the SentenceTransformer model
print(f"🔄 Loading SentenceTransformer model from {MODEL_DIR}...")
model = SentenceTransformer(MODEL_DIR)

# ✅ Ensure model is in eval mode
model.eval()

# ✅ Create dummy input sentence & tokenize it
dummy_text = ["This is a test sentence."]
inputs = model.tokenize(dummy_text)

# ✅ Convert input tensors to PyTorch format
input_ids = torch.tensor(inputs["input_ids"])
attention_mask = torch.tensor(inputs["attention_mask"])

# ✅ Export model to ONNX
torch.onnx.export(
    model,
    (input_ids, attention_mask),  # ✅ Pass correct inputs
    ONNX_MODEL_PATH,
    export_params=True,
    opset_version=13,
    do_constant_folding=True,
    input_names=["input_ids", "attention_mask"],  # ✅ Match input format
    output_names=["sentence_embedding"],
    dynamic_axes={
        "input_ids": {0: "batch_size", 1: "sequence_length"},
        "attention_mask": {0: "batch_size", 1: "sequence_length"},
        "sentence_embedding": {0: "batch_size", 1: "embedding_size"},
    },
)

print(f"✅ ONNX conversion completed successfully! Model saved at {ONNX_MODEL_PATH}")
