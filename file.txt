import torch
import numpy as np
import onnx
import onnxruntime
from sentence_transformers import SentenceTransformer

# ✅ Define paths
MODEL_DIR = "c:/users/abc/downloads/code-baton/jina-embedding-s-en-v1"  # Path to downloaded model
ONNX_MODEL_PATH = "c:/users/abc/downloads/code-baton/sentence_transformer.onnx"

# ✅ Load SentenceTransformer model
print(f"🔄 Loading SentenceTransformer model from {MODEL_DIR}...")
model = SentenceTransformer(MODEL_DIR)

# ✅ Create a dummy input sentence
dummy_text = ["This is a test sentence."]
dummy_embedding = model.encode(dummy_text, convert_to_tensor=True)

# ✅ Convert dummy input to numpy array
input_ids = torch.tensor(dummy_embedding.cpu().numpy(), dtype=torch.float32)

# ✅ Ensure model is in eval mode
model.eval()

# ✅ Convert model to ONNX
torch.onnx.export(
    model,
    input_ids,  # Correct input format
    ONNX_MODEL_PATH,
    export_params=True,
    opset_version=13,
    do_constant_folding=True,
    input_names=["input_ids"],
    output_names=["sentence_embedding"],
    dynamic_axes={"input_ids": {0: "batch_size", 1: "embedding_size"}},
)

print(f"✅ ONNX conversion completed successfully! Model saved at {ONNX_MODEL_PATH}")
