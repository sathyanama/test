def correct_text(input_text):
    prompt = f"""[INST] <<SYS>> 
Correct the mistakes, grammar, and punctuation in the text. 
Only return the corrected text without any explanations. 
Do not repeat the input text. 
<</SYS>> 

{input_text} [/INST]"""

    tokens_to_generate = min(2048, len(input_text.split()) * 4)  # Increased for full response
    avg_time_per_token = 0.04 if torch.cuda.is_available() else 0.08
    estimated_total_time = tokens_to_generate * avg_time_per_token

    print(f"\nEstimated response time: {estimated_total_time:.1f} seconds.")
    stop_spinner.clear()
    spinner_thread = threading.Thread(target=spinner, args=(estimated_total_time,))
    spinner_thread.start()

    try:
        start_time = time.time()
        outputs = pipeline(
            prompt,
            max_new_tokens=tokens_to_generate,
            eos_token_id=tokenizer.eos_token_id,
            do_sample=True,
            temperature=model.config.temperature,
            top_p=model.config.top_p,
            num_beams=2,
            repetition_penalty=model.config.repetition_penalty,
            early_stopping=True  # Ensures full response generation
        )

        stop_spinner.set()
        spinner_thread.join()

        actual_time_taken = time.time() - start_time

        # ✅ Extract only the longest, most complete response
        corrected_text = outputs[0]["generated_text"].strip()
        corrected_text = clean_output(corrected_text)

        # ✅ Ensure we take only the final corrected version, removing any duplicates
        corrected_text_lines = corrected_text.split("\n")
        unique_lines = list(dict.fromkeys(corrected_text_lines))  # Removes duplicate lines while maintaining order
        corrected_text = "\n".join(unique_lines).strip()

        print(f"\nProcessed in {actual_time_taken:.1f} seconds.")
        return corrected_text

    except KeyboardInterrupt:
        stop_spinner.set()
        spinner_thread.join()
        print("\nProcess interrupted by user (Ctrl + C). Exiting gracefully.\n")
        sys.exit(0)
