//OPT

private Map<String, Object> buildHandoverMapOPT(String team,
                                                LocalDateTime shiftStart,
                                                LocalDateTime shiftEnd,
                                                String heading) {

    final LocalDateTime finalShiftStart = shiftStart;
    final LocalDateTime finalShiftEnd = shiftEnd;

    // Current shift HO
    Optional<HandoverTracker> handoverOptional =
            handoverService.getHandoverByShift(team, finalShiftStart, finalShiftEnd);

    HandoverTracker handover = handoverOptional.orElseGet(() -> {
        HandoverTracker newHandover = new HandoverTracker();
        newHandover.setStartDate(finalShiftStart);
        newHandover.setEndDate(finalShiftEnd);
        newHandover.setHoItem("");
        return newHandover;
    });

    // Fallback → previous valid HO
    if (handoverOptional.isEmpty() ||
        handover.getHoItem() == null ||
        handover.getHoItem().isBlank()) {

        Optional<HandoverTracker> fallbackHO =
                handoverService.getLatestHandoverBefore(team, finalShiftStart);

        fallbackHO.ifPresent(fb -> {
            if (fb.getHoItem() != null && !fb.getHoItem().isBlank()) {
                handover.setHoItem(fb.getHoItem());
                handover.setStartDate(fb.getStartDate());
                handover.setEndDate(fb.getEndDate());
            }
        });
    }

    return Map.of("heading", heading, "handover", handover);
}
