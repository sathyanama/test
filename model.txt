from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from transformers import AutoTokenizer, AutoModelForSeq2SeqLM
import torch

app = FastAPI()

# Load Flan-T5-Base model and tokenizer
MODEL_PATH = "plan/to/flan-t5-base"  # Update this with your actual path
tokenizer = AutoTokenizer.from_pretrained(MODEL_PATH)
model = AutoModelForSeq2SeqLM.from_pretrained(MODEL_PATH).to("cuda" if torch.cuda.is_available() else "cpu")

class InputText(BaseModel):
    text: str

def chunk_text(text, max_tokens=512):
    """Splits long text into smaller chunks of max_tokens each."""
    words = text.split()
    return [" ".join(words[i : i + max_tokens]) for i in range(0, len(words), max_tokens)]

@app.post("/revise-text/")
def revise_text(request: InputText):
    input_text = request.text.strip()

    if not input_text:
        raise HTTPException(status_code=400, detail="Input text cannot be empty.")

    # Split long text into manageable chunks
    text_chunks = chunk_text(input_text, max_tokens=512)
    
    revised_chunks = []
    for chunk in text_chunks:
        inputs = tokenizer(chunk, return_tensors="pt", truncation=True, padding="longest", max_length=512)
        inputs = {k: v.to(model.device) for k, v in inputs.items()}
        
        outputs = model.generate(**inputs, max_length=512, num_beams=5, early_stopping=True)
        revised_chunks.append(tokenizer.decode(outputs[0], skip_special_tokens=True))
    
    revised_text = " ".join(revised_chunks)  # Merge all revised chunks back together
    return {"original_text": input_text, "revised_text": revised_text}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
