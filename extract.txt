final_contents = []

for c in result:              # result from asyncio.run(run())
    for p in c.parts:         # each Part(text=...)
        txt = p.text.strip()
        
        # Case 1: Fetcher output (status + data + result)
        if "status" in txt and "data" in txt and "result" in txt:
            try:
                parsed = json.loads(txt.replace("'", '"'))  # normalize quotes
                final_contents.append({
                    "status": parsed.get("status"),
                    "results": parsed.get("data", {}).get("results")
                })
            except Exception:
                final_contents.append(txt)  # fallback if parsing fails

        # Case 2: Summarizer output (topic + summary sentences)
        elif "topic" in txt and "Summary_sentence_1" in txt:
            try:
                parsed = json.loads(txt)
                summary_only = {
                    "topic": parsed.get("topic")
                }
                # dynamically pull all summary sentences
                for k, v in parsed.items():
                    if k.lower().startswith("summary_sentence"):
                        summary_only[k] = v
                final_contents.append(summary_only)
            except Exception:
                final_contents.append(txt)  # fallback
