from transformers import AutoTokenizer, AutoModelForCausalLM
import torch

# 🔹 Change this to your office model path
MODEL_PATH = "C:/models/mistral-7b/"  # Update with actual path where `part_aa` files are stored

# Load Tokenizer
tokenizer = AutoTokenizer.from_pretrained(MODEL_PATH)

# Load Model (without index.json)
model = AutoModelForCausalLM.from_pretrained(
    MODEL_PATH, 
    torch_dtype=torch.float16,  # Use FP16 for performance
    device_map="auto"  # Automatically use CPU or GPU
)

# 🔹 Test with Example Input
input_text = "Correct the grammar in this sentence: i wet to ofice and not able to meet boss"
inputs = tokenizer(input_text, return_tensors="pt").to("cuda" if torch.cuda.is_available() else "cpu")

# Generate Output
output = model.generate(**inputs, max_new_tokens=100)
revised_text = tokenizer.decode(output[0], skip_special_tokens=True)

print("🔹 Revised Text:", revised_text)
