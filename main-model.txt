# main_menu.py
import importlib
import sys

mistral_loaded = False
meta_loaded = False
mistral_module = None
meta_module = None

def load_mistral():
    global mistral_loaded, mistral_module
    if not mistral_loaded:
        print("\n🔄 Loading Mistral model...")
        mistral_module = importlib.import_module("mistral_module")
        mistral_loaded = True
        print("✅ Mistral model loaded.\n")
    mistral_module.run_mistral_revision()

def load_meta():
    global meta_loaded, meta_module
    if not meta_loaded:
        print("\n🔄 Loading Meta-LLaMA model...")
        meta_module = importlib.import_module("meta_module")
        meta_loaded = True
        print("✅ Meta-LLaMA model loaded.\n")
    meta_module.run_meta_revision()

while True:
    print("=== AI Comm Revision Menu ===")
    print("1. Mistral Technical Comm Revision")
    print("2. Meta-LLaMA Revision")
    print("3. Quit")

    choice = input("\nChoose an option [1/2/3]: ").strip()

    if choice == "1":
        load_mistral()
    elif choice == "2":
        load_meta()
    elif choice == "3":
        print("👋 Exiting. Goodbye!")
        sys.exit()
    else:
        print("❌ Invalid option. Try again.\n")
