def summarize_logs(logs):
    if not logs:
        return "No logs found."

    # Count http codes
    http_2xx = sum(log["count"] for log in logs if 200 <= log["httpCode"] < 300)
    http_4xx = sum(log["count"] for log in logs if 400 <= log["httpCode"] < 500)
    http_5xx = sum(log["count"] for log in logs if 500 <= log["httpCode"] < 600)

    # Bypass model if only 2xx codes exist
    if http_4xx == 0 and http_5xx == 0:
        return (
            f"No 4xx or 5xx errors were found in the logs.\n"
            f"There were a total of {http_2xx} successful (2xx) responses.\n"
            f"System appears healthy, no further action is needed."
        )

    # Else, use LLM for detailed error summary
    formatted_logs = "\n".join([str(log) for log in logs])
    prompt = f"""
You are a Splunk log analyst.

Below are sample log records from a Splunk query:
{formatted_logs}

Instructions:
1. If there are any 4xx or 5xx codes, identify the top 2 error types by count.
2. Count how many responses occurred for 2xx, 4xx, and 5xx status ranges.
3. Mention 5xx errors **only if they are present in the logs**.
4. Do not invent or guess any errors that are not present.
5. If no 4xx or 5xx are found, clearly say: "No 4xx or 5xx errors were found."
6. Do NOT repeat these instructions or format headings. Return only the final summary.

Return only the summary in clean English.
"""
    return mistral_generate(prompt, max_tokens=512)
