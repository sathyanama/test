package com.example.demo.service;

import com.example.demo.config.AdfsProperties;
import com.example.demo.model.TicketDetails;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.MockitoAnnotations;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.client.RestTemplate;

import java.lang.reflect.Field;
import java.util.Collections;
import java.util.List;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

public class TicketServiceTest {

    @Mock
    private AdfsProperties adfsProperties;

    @InjectMocks
    private TicketService ticketService;

    private RestTemplate mockTemplate;

    @BeforeEach
    public void setup() throws Exception {
        MockitoAnnotations.openMocks(this);
        mockTemplate = mock(RestTemplate.class);
        Field restTemplateField = TicketService.class.getDeclaredField("restTemplate");
        restTemplateField.setAccessible(true);
        restTemplateField.set(ticketService, mockTemplate);
    }

    @Test
    public void testGetT3IncidentDetails_successful() throws InterruptedException {
        String incidentNumber = "INC123";
        String externalUrl = "api to be added";

        when(adfsProperties.getExternalT3URL()).thenReturn(externalUrl);

        String json = "[{" +
                "\"cgsincident.incident_numner\": \"INC123\"," +
                "\"cgsincident.short_description\": \"Network Down\"," +
                "\"cgsincident.state\": \"Open\"," +
                "\"cgsincident.assignment_group\": \"NetOps\"," +
                "\"cgsincident.assignment_To\": \"Alice\"," +
                "\"cgsincident.impact\": \"High\"," +
                "\"cgsincident.urgency\": \"Critical\"," +
                "\"cgsincident.priority\": \"P1\"," +
                "\"cgsincident.opened_time_est\": \"2024-04-01 10:00:00\"," +
                "\"cgsincident.closed_time_est\": \"2024-04-01 12:00:00\"," +
                "\"cgsincident.actual_impact_classification\": \"Critical\"," +
                "\"cgsincident.classification\": \"Service\"," +
                "\"cgsincident.impacted_lob\": \"Banking\"," +
                "\"cgsincident.parent_incident\": \"INC0001\"," +
                "\"cgsincident.category\": \"Network\"," +
                "\"cgsincident.sub_category\": \"Router\"," +
                "\"cgsincident.owning_lob_level4\": \"Infra\"," +
                "\"cgsincident.change_request\": \"CHG001\"," +
                "\"cgsincident.external_vendor\": \"VendorA\"," +
                "\"cgsincident.opened_by\": \"system.user\"}]";

        ResponseEntity<String> responseEntity = new ResponseEntity<>(json, HttpStatus.OK);

        when(mockTemplate.getForEntity(eq(externalUrl + incidentNumber), eq(String.class)))
                .thenReturn(responseEntity);

        List<TicketDetails> result = ticketService.getT3IncidentDetails(incidentNumber);

        assertEquals(1, result.size());
        TicketDetails ticket = result.get(0);
        assertEquals("INC123", ticket.getIncidentNumber());
        assertEquals("Network Down", ticket.getShortDescription());
        assertEquals("Open", ticket.getState());
        assertEquals("NetOps", ticket.getAssignmentGroup());
        assertEquals("Alice", ticket.getAssignmentTo());
        assertEquals("High", ticket.getImpact());
        assertEquals("Critical", ticket.getUrgency());
        assertEquals("P1", ticket.getPriority());
        assertEquals("2024-04-01 10:00:00", ticket.getOpenedTime());
        assertEquals("2024-04-01 12:00:00", ticket.getClosedTime());
        assertEquals("Critical", ticket.getActImpactClassification());
        assertEquals("Service", ticket.getClassification());
        assertEquals("Banking", ticket.getImpactedLob());
        assertEquals("INC0001", ticket.getParentIncident());
        assertEquals("Network", ticket.getCategory());
        assertEquals("Router", ticket.getSubCategory());
        assertEquals("Infra", ticket.getOwningLobLevel4());
        assertEquals("CHG001", ticket.getChangeRequest());
        assertEquals("VendorA", ticket.getExternalVendor());
        assertEquals("system.user", ticket.getCreateBy());
    }

    @Test
    public void testGetT3IncidentDetails_nonOKResponse() throws InterruptedException {
        String incidentNumber = "INC404";
        String externalUrl = "api to be added";

        when(adfsProperties.getExternalT3URL()).thenReturn(externalUrl);
        when(mockTemplate.getForEntity(eq(externalUrl + incidentNumber), eq(String.class)))
                .thenReturn(new ResponseEntity<>("", HttpStatus.INTERNAL_SERVER_ERROR));

        List<TicketDetails> result = ticketService.getT3IncidentDetails(incidentNumber);
        assertNotNull(result);
        assertTrue(result.isEmpty());
    }

    @Test
    public void testGetT3IncidentDetails_exceptionHandling() throws InterruptedException {
        String incidentNumber = "INC500";
        String externalUrl = "api to be added";

        when(adfsProperties.getExternalT3URL()).thenReturn(externalUrl);
        when(mockTemplate.getForEntity(eq(externalUrl + incidentNumber), eq(String.class)))
                .thenThrow(new RuntimeException("Simulated failure"));

        List<TicketDetails> result = ticketService.getT3IncidentDetails(incidentNumber);
        assertNotNull(result);
        assertTrue(result.isEmpty());
    }

    @Test
    public void testGetT3IncidentDetails_emptyJson() throws InterruptedException {
        String incidentNumber = "INC000";
        String externalUrl = "api to be added";

        when(adfsProperties.getExternalT3URL()).thenReturn(externalUrl);
        when(mockTemplate.getForEntity(eq(externalUrl + incidentNumber), eq(String.class)))
                .thenReturn(new ResponseEntity<>("[]", HttpStatus.OK));

        List<TicketDetails> result = ticketService.getT3IncidentDetails(incidentNumber);
        assertNotNull(result);
        assertTrue(result.isEmpty());
    }

    @Test
    public void testGetT3IncidentDetails_malformedJson() throws InterruptedException {
        String incidentNumber = "INC999";
        String externalUrl = "api to be added";

        when(adfsProperties.getExternalT3URL()).thenReturn(externalUrl);
        when(mockTemplate.getForEntity(eq(externalUrl + incidentNumber), eq(String.class)))
                .thenReturn(new ResponseEntity<>("[{invalid_json}]", HttpStatus.OK));

        List<TicketDetails> result = ticketService.getT3IncidentDetails(incidentNumber);
        assertNotNull(result);
        assertTrue(result.isEmpty());
    }
}
