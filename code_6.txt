import time
from contextvars import ContextVar

session_service_ctx = ContextVar("session_service", default=None)
last_refresh_ctx = ContextVar("last_refresh", default=0)
REFRESH_INTERVAL = 25 * 60  # 25 minutes (refresh before Oracle idle timeout)


def get_session_service():
    """Return a valid, per-request DatabaseSessionService (auto-refresh + safe fallback)."""
    db_session = session_service_ctx.get()
    last_refresh = last_refresh_ctx.get()
    now = time.time()

    # Decide whether we need to create or refresh the connection
    if db_session is None or now - last_refresh > REFRESH_INTERVAL:
        print("Creating/refreshing Oracle session for this request")
        try:
            db_session = DatabaseSessionService(
                db_url=db_url,
                connect_args={
                    "host": host,
                    "port": port,
                    "service_name": service_name,
                },
            )
            print("Database session refreshed successfully.")
            session_service_ctx.set(db_session)
            last_refresh_ctx.set(now)

        except Exception as e:
            # Handles ORA-03113, DPI-1080, Kerberos failures, etc.
            print(f"Database session refresh failed: {e}")
            print("Falling back to InMemorySessionService for this request.")
            db_session = InMemorySessionService()
            session_service_ctx.set(db_session)
            last_refresh_ctx.set(now)

    return db_session
