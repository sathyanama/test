

@Query(value = "SELECT * FROM incident_tracker i " +
               "WHERE UPPER(i.incident_number) LIKE UPPER(CONCAT('%', :query, '%')) " +
               "OR UPPER(i.customer_exp) LIKE UPPER(CONCAT('%', :query, '%')) " +
               "OR UPPER(i.analysis) LIKE UPPER(CONCAT('%', :query, '%')) " +
               "OR UPPER(i.alert_name) LIKE UPPER(CONCAT('%', :query, '%')) " +
               "ORDER BY i.create_datetime DESC " +
               "FETCH FIRST 100 ROWS ONLY",
       nativeQuery = true)
List<ChatIncident> fetchIncidentsJava(@Param("query") String query);




@Query(value = "SELECT * FROM incident_tracker i " +
               "WHERE (" +
               "UPPER(i.incident_number) LIKE UPPER('%' || :word1 || '%') " +
               "OR UPPER(DBMS_LOB.SUBSTR(i.customer_exp, 4000, 1)) LIKE UPPER('%' || :word1 || '%') " +
               "OR UPPER(DBMS_LOB.SUBSTR(i.analysis, 4000, 1)) LIKE UPPER('%' || :word1 || '%') " +
               "OR UPPER(DBMS_LOB.SUBSTR(i.alert_name, 4000, 1)) LIKE UPPER('%' || :word1 || '%') " +
               "OR UPPER(i.incident_number) LIKE UPPER('%' || :word2 || '%') " +
               "OR UPPER(DBMS_LOB.SUBSTR(i.customer_exp, 4000, 1)) LIKE UPPER('%' || :word2 || '%') " +
               "OR UPPER(DBMS_LOB.SUBSTR(i.analysis, 4000, 1)) LIKE UPPER('%' || :word2 || '%') " +
               "OR UPPER(DBMS_LOB.SUBSTR(i.alert_name, 4000, 1)) LIKE UPPER('%' || :word2 || '%')" +
               ") " +
               "ORDER BY i.create_datetime DESC " +
               "FETCH FIRST 100 ROWS ONLY",
       nativeQuery = true)
List<ChatIncident> fetchIncidentsSql(@Param("word1") String word1, @Param("word2") String word2);
