@Test
    void test_viewByFolder_withSuccessMessage() throws Exception {
        Long folderId = 1L;
        int page = 0;

        List<RcaDetails> dummyRecords = List.of(new RcaDetails());
        Page<RcaDetails> mockPage = new PageImpl<>(dummyRecords);
        RcaFolder folder = new RcaFolder();
        folder.setFolderName("Test Folder");

        when(rcaService.getDetailsByFolder(folderId, page, 10)).thenReturn(mockPage);
        when(rcaService.getFolderById(folderId)).thenReturn(Optional.of(folder));

        MockHttpSession session = new MockHttpSession();
        session.setAttribute("username", "sathya");
        session.setAttribute("team", "DMC");
        session.setAttribute("role", "User");

        mockMvc.perform(get("/view")
                .param("folderId", folderId.toString())
                .param("page", String.valueOf(page))
                .param("status", "success")
                .param("message", "Loaded successfully")
                .session(session))
            .andExpect(status().isOk())
            .andExpect(view().name("rcaView"))
            .andExpect(model().attribute("username", "sathya"))
            .andExpect(model().attribute("team", "DMC"))
            .andExpect(model().attribute("role", "User"))
            .andExpect(model().attribute("records", dummyRecords))
            .andExpect(model().attribute("currentPage", page))
            .andExpect(model().attribute("totalPages", 1))
            .andExpect(model().attribute("folderId", folderId))
            .andExpect(model().attribute("folderName", "Test Folder"))
            .andExpect(model().attribute("successMessage", "Loaded successfully"));
    }
