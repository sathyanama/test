import os
import time
import functools
import inspect
import logging

def log_execution_time(logger: logging.Logger):
    def decorator(func):
        if inspect.iscoroutinefunction(func):
            # --- Async wrapper ---
            @functools.wraps(func)
            async def wrapper(*args, **kwargs):
                request_id = f"req-{int(time.time())}-{os.urandom(4).hex()}"
                start_time = time.time()
                try:
                    result = await func(*args, **kwargs)   # await async route
                    end_time = time.time()
                    execution_time = end_time - start_time
                    logger.info("[%s] %s took %.2f ms", request_id, func.__name__, execution_time * 1000)
                    return result
                except Exception as e:
                    end_time = time.time()
                    execution_time = end_time - start_time
                    logger.error("[%s] %s failed after %.2f ms: %s", request_id, func.__name__, execution_time * 1000, str(e))
                    raise
            return wrapper
        else:
            # --- Sync wrapper ---
            @functools.wraps(func)
            def wrapper(*args, **kwargs):
                request_id = f"req-{int(time.time())}-{os.urandom(4).hex()}"
                start_time = time.time()
                try:
                    result = func(*args, **kwargs)   # sync call
                    end_time = time.time()
                    execution_time = end_time - start_time
                    logger.info("[%s] %s took %.2f ms", request_id, func.__name__, execution_time * 1000)
                    return result
                except Exception as e:
                    end_time = time.time()
                    execution_time = end_time - start_time
                    logger.error("[%s] %s failed after %.2f ms: %s", request_id, func.__name__, execution_time * 1000, str(e))
                    raise
            return wrapper
    return decorator
