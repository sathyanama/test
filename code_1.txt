CREATE OR REPLACE TRIGGER sessions_ignore_dup_trg
BEFORE INSERT ON sessions
FOR EACH ROW
DECLARE
    v_exists NUMBER;
BEGIN
    -- Check if a session row already exists for same ID
    SELECT COUNT(*) INTO v_exists
      FROM sessions
     WHERE id = :NEW.id;

    IF v_exists > 0 THEN
        -- Prevent duplicate insert silently
        raise_application_error(-20999, 'DUPLICATE_SESSION');
    END IF;

EXCEPTION
    WHEN OTHERS THEN
        IF SQLCODE = -20999 THEN
            NULL;  -- ignore duplicate
        ELSE
            RAISE;
        END IF;
END;
/
