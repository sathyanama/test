import sys
from transformers import T5Tokenizer, T5ForConditionalGeneration

# Load the Flan-T5-Base model and tokenizer
model_name = "google/flan-t5-base"
tokenizer = T5Tokenizer.from_pretrained(model_name)
model = T5ForConditionalGeneration.from_pretrained(model_name)

def correct_text(text):
    """Fixes grammar, spelling, and pronunciation issues in the given text."""
    prompt = f"Correct the grammar, spelling, and pronunciation: {text}"
    input_ids = tokenizer(prompt, return_tensors="pt").input_ids
    output_ids = model.generate(input_ids, max_length=512, num_beams=5, early_stopping=True)
    corrected_text = tokenizer.decode(output_ids[0], skip_special_tokens=True)
    return corrected_text

def main():
    print("Enter your text (Press Enter twice to submit or type 'exit' to quit):")
    
    # Read multiple lines of input until an empty line is encountered
    input_text = []
    while True:
        try:
            line = input()
            if line.strip().lower() == "exit":  # Exit condition
                print("Exiting program...")
                sys.exit(0)
            if line.strip() == "":
                break
            input_text.append(line)
        except EOFError:
            break

    if not input_text:
        print("No input provided. Exiting...")
        return

    # Join all lines into a single paragraph
    full_text = " ".join(input_text)

    # Generate and display the corrected text
    corrected_output = correct_text(full_text)
    print("\n=== Corrected Text ===\n")
    print(corrected_output)

if __name__ == "__main__":
    main()
