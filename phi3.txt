import transformers
import torch
import time
import sys
import os
import signal
import re

# Fix SIGALRM Error on Windows
if os.name == "nt":
    if hasattr(signal, "SIGALRM"):
        del signal.SIGALRM  # Prevents AttributeError on Windows

# Load Model from Local Path (No Internet Required)
model_id = "Phi-3-mini-128k-instruct/"

print("INFO: Loading tokenizer (Offline Mode)...")
tokenizer = transformers.AutoTokenizer.from_pretrained(
    model_id,
    local_files_only=True,  # Ensures no online download
    trust_remote_code=True  # Fixes special token issue
)
print("INFO: Tokenizer loaded successfully.")

print("INFO: Loading model (Offline Mode)... (This may take time)")
try:
    model = transformers.AutoModelForCausalLM.from_pretrained(
        model_id,
        torch_dtype=torch.float32 if torch.cuda.is_available() else torch.bfloat16,
        device_map="auto",
        low_cpu_mem_usage=True,
        local_files_only=True,  # Ensures no online download
        trust_remote_code=True  # Fixes custom code execution issue
    )
    print("INFO: Model loaded successfully.")
except Exception as e:
    print(f"ERROR: Model loading failed: {e}")
    sys.exit(1)

# Create a text generation pipeline
print("INFO: Initializing text generation pipeline...")
pipeline = transformers.pipeline(
    "text-generation",
    model=model,
    tokenizer=tokenizer,
)
print("INFO: Text generation pipeline is ready.")

# Function to Clean the Output and Extract Only the Corrected Text
def clean_output(output_text):
    output_text = re.sub(r"\[INST\].*?\[/INST\]", "", output_text, flags=re.DOTALL).strip()
    output_text = re.sub(r"<<SYS>>.*?<</SYS>>", "", output_text, flags=re.DOTALL).strip()
    output_text = re.sub(r"\[/SYS\]", "", output_text).strip()
    return output_text

# Function to Process Text Correction
def correct_text(input_text):
    print("INFO: Generating corrected text...")
    prompt = f"[INST] <<SYS>> Correct the mistakes, grammar, and punctuation in the text. Only return the corrected text without any explanations. <</SYS>>\n{input_text} [/INST]"

    tokens_to_generate = min(300, len(input_text.split()) * 2)

    try:
        start_time = time.time()
        outputs = pipeline(
            prompt,
            max_new_tokens=tokens_to_generate,
        )
        actual_time_taken = time.time() - start_time

        corrected_text = outputs[0]["generated_text"].strip()
        corrected_text = clean_output(corrected_text)

        print(f"INFO: Text generation completed in {actual_time_taken:.1f} seconds.")
        return corrected_text

    except Exception as e:
        print(f"ERROR: Text generation failed: {e}")
        return ""

# Run Correction
while True:
    input_text = input("\nEnter text for correction (type 'quit' to exit): ")
    if input_text.lower() == "quit":
        print("Exiting program. Goodbye.")
        break

    corrected_text = correct_text(input_text)
    if corrected_text:
        print("\nCorrected Text:\n" + corrected_text)
