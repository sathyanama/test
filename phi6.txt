import transformers
import torch
import time
import sys
import os
import signal
import re
import threading
from itertools import cycle

# Fix SIGALRM Error on Windows
if os.name == "nt":
    if hasattr(signal, "SIGALRM"):
        del signal.SIGALRM  # Prevents AttributeError on Windows

# Load Model from Local Path (No Internet Required)
model_id = "Phi-3-mini-128k-instruct/"

# Load Tokenizer (Ensure no download, allow remote code execution)
tokenizer = transformers.AutoTokenizer.from_pretrained(
    model_id,
    local_files_only=True,  
    trust_remote_code=True  
)

# Load Model (Force Load Phi3ForCausalLM for local compatibility)
try:
    model = transformers.AutoModelForCausalLM.from_pretrained(
        model_id,
        torch_dtype=torch.float32 if torch.cuda.is_available() else torch.bfloat16,
        device_map="auto",
        low_cpu_mem_usage=True,
        local_files_only=True,  
        trust_remote_code=True  
    )

    # 🔹 Fix: Explicitly set configuration to prevent modification warnings
    model.config.use_cache = True
    model.config.max_new_tokens = 512  # Increased from 300 to prevent truncation
    model.config.temperature = 0.7
    model.config.top_p = 0.95
    model.config.repetition_penalty = 1.2

except Exception as e:
    print(f"ERROR: Model loading failed: {e}")
    sys.exit(1)

# Create a text generation pipeline
pipeline = transformers.pipeline(
    "text-generation",
    model=model,
    tokenizer=tokenizer,
)

# Function to show a loading spinner while generating text
stop_spinner = threading.Event()

def spinner(estimated_time):
    start_time = time.time()
    for symbol in cycle(["⠋", "⠙", "⠹", "⠸", "⠼", "⠴", "⠦", "⠧", "⠇", "⠏"]):
        elapsed_time = time.time() - start_time
        remaining_time = max(estimated_time - elapsed_time, 0)
        if stop_spinner.is_set():
            break
        print(f"\rProcessing... {symbol} | Estimated time left: {remaining_time:.1f}s", end="", flush=True)
        time.sleep(0.2)

# Function to clean the output and extract only the corrected text
def clean_output(output_text):
    output_text = re.sub(r"\[INST\].*?\[/INST\]", "", output_text, flags=re.DOTALL).strip()
    output_text = re.sub(r"<<SYS>>.*?<</SYS>>", "", output_text, flags=re.DOTALL).strip()
    output_text = re.sub(r"\[/SYS\]", "", output_text).strip()
    return output_text

# Function to process text correction with spinner and estimated time
def correct_text(input_text):
    prompt = f"[INST] <<SYS>> Correct the mistakes, grammar, and punctuation in the text. Only return the corrected text without any explanations. <</SYS>>\n{input_text} [/INST]"

    tokens_to_generate = min(512, len(input_text.split()) * 2)  # Increased token limit
    avg_time_per_token = 0.04 if torch.cuda.is_available() else 0.08  # Estimate time based on device speed
    estimated_total_time = tokens_to_generate * avg_time_per_token

    print(f"Estimated response time: {estimated_total_time:.1f} seconds.")

    stop_spinner.clear()
    spinner_thread = threading.Thread(target=spinner, args=(estimated_total_time,))
    spinner_thread.start()

    try:
        start_time = time.time()
        outputs = pipeline(
            prompt,
            max_new_tokens=tokens_to_generate,
        )
        stop_spinner.set()
        spinner_thread.join()

        actual_time_taken = time.time() - start_time
        corrected_text = outputs[0]["generated_text"].strip()
        corrected_text = clean_output(corrected_text)

        print(f"\nProcessing completed in {actual_time_taken:.1f} seconds.")
        return corrected_text

    except Exception as e:
        stop_spinner.set()
        spinner_thread.join()
        print(f"ERROR: Text generation failed: {e}")
        return ""

# Run Correction with Multi-Line Input (Minimal Output)
try:
    while True:
        print("\nEnter text for correction (Paste multi-line text and press Enter twice to process, type 'quit' to exit):")

        lines = []
        while True:
            try:
                line = input()
                if line.strip().lower() == "quit":
                    print("Exiting program.")
                    sys.exit(0)
                if line.strip() == "":
                    break  
                lines.append(line)
            except EOFError:
                break

        input_text = "\n".join(lines).strip()

        if input_text:
            corrected_text = correct_text(input_text)
            print("\n" + corrected_text + "\n")

except KeyboardInterrupt:
    stop_spinner.set()
    print("\nExiting program safely.")
    sys.exit(0)
