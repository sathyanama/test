 @JsonFormat(pattern = "yyyy-MM-dd'T'HH:mm:ss")
 <span th:text="${incident.startDatetimeFormatted}"></span>
 
 
public String getStartDatetimeFormatted() {
    try {
        return startDatetime != null
            ? startDatetime.format(DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm"))
            : "";
    } catch (Exception e) {
        return "";
    }
}

		
<input type="hidden"
       name="startDatetime"
       th:value="${incident.startDatetimeFormatted}" />
