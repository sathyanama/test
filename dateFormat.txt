private static String formatTimestamps(List<String> timestamps) {
    DateTimeFormatter inputFormat = DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm"); // adjust if needed
    return timestamps.stream()
        .map(ts -> {
            try {
                LocalDateTime dateTime = LocalDateTime.parse(ts, inputFormat);
                return dateTime.format(inputFormat);  // keep consistent format
            } catch (Exception e) {
                return ts;  // fallback: return raw if parsing fails
            }
        })
        .collect(Collectors.joining(","));
}
