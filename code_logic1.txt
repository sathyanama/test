import time
from smart_sdk.runners import DatabaseSessionService, InMemorySessionService

# keep your existing context vars and constants
# session_service_ctx, last_refresh_ctx, REFRESH_INTERVAL already defined
_shared_session = None
_shared_last_refresh = 0.0


def get_session_service():
    """Return a stable, pod-level DatabaseSessionService with auto-refresh."""
    global _shared_session, _shared_last_refresh

    db_session = session_service_ctx.get()
    last_refresh = last_refresh_ctx.get()
    now = time.time()

    current_krb_conf = os.environ.get("KRB5_CONFIG")
    current_krb_cache = os.environ.get("KRB5CCNAME")

    set_kerberos_config()

    try:
        # Use pod-shared session if local context is empty
        if db_session is None:
            db_session = _shared_session

        # Create or refresh if idle too long
        if db_session is None or now - _shared_last_refresh > REFRESH_INTERVAL:
            print("Creating/refreshing Oracle session for this pod")
            db_session = DatabaseSessionService(
                db_url=db_url,
                connect_args={
                    "host": host,
                    "port": port,
                    "service_name": service_name,
                },
            )
            _shared_session = db_session
            _shared_last_refresh = now
            session_service_ctx.set(db_session)
            last_refresh_ctx.set(now)
            print("Database session refreshed successfully.")

    except Exception as e:
        print(f"️Database session refresh failed: {e}")
        print("Falling back to InMemorySessionService.")
        db_session = InMemorySessionService()
        _shared_session = db_session
        _shared_last_refresh = now
        session_service_ctx.set(db_session)
        last_refresh_ctx.set(now)

    # restore proxy/kerberos settings
    if current_krb_conf:
        os.environ["KRB5_CONFIG"] = current_krb_conf
    if current_krb_cache:
        os.environ["KRB5CCNAME"] = current_krb_cache

    return db_session
