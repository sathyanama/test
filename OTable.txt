CREATE TABLE incident_tracker (
    ID NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    INCIDENT_NUMBER VARCHAR2(255) NOT NULL,
    CREATE_BY VARCHAR2(255) NOT NULL,
    CREATE_DATETIME TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    TEAM VARCHAR2(255) NOT NULL,
    START_DATETIME TIMESTAMP NULL,
    DEDUCT_DATETIME TIMESTAMP NULL,
    DIAGNOSE_DATETIME TIMESTAMP NULL,
    MITIGATE_DATETIME TIMESTAMP NULL,
    RESOLVE_DATETIME TIMESTAMP NULL,
    MC_ENGAGE_DATETIME TIMESTAMP NULL,
    CUSTOMER_EXP VARCHAR2(255) NOT NULL,
    ANALYSIS CLOB NULL,
    IMPACT_COUNT NUMBER NULL,
    FLOW VARCHAR2(500) NULL,
    UPDATE_USER VARCHAR2(255) NOT NULL,
    UPDATE_DATETIME TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    ALERT_NAME VARCHAR2(255) NULL,
    FLAG_ITEM NUMBER(1) DEFAULT 0 NULL,
    PARENT_INCIDENT VARCHAR2(50) NULL,
    ASSIGNMENT_GROUP VARCHAR2(500) NULL,
    ASSIGNED_TO VARCHAR2(500) NULL,
    OPENED_TIME_EST TIMESTAMP NULL,
    CLOSED_TIME_EST TIMESTAMP NULL,
    CUSTOMER_IMPACT CLOB NULL,
    SMT_SPLIT_DARK_CANARY_ATC VARCHAR2(2000) NULL,
    FOLLOW_UP_ITEMS CLOB NULL,
    TEAMS_ENGAGED VARCHAR2(2000) NULL,
    SPLUNK_QUERIES CLOB NULL,
    CUSTOMER_USER_JOURNEY VARCHAR2(1) NULL,
    DASHBOARD CLOB NULL,
    OWNING_LOB_LEVEL_4 VARCHAR2(1000) NULL,
    CATEGORY VARCHAR2(500) NULL,
    SUBCATEGORY VARCHAR2(500) NULL,
    TAG VARCHAR2(200) NULL,
    ACT_IMP_CLASSIFICATION VARCHAR2(500) NULL,
    CLASSIFICATION VARCHAR2(500) NULL,
    IMPACT VARCHAR2(100) NULL,
    URGENCY VARCHAR2(100) NULL,
    PRIORITY VARCHAR2(100) NULL,
    IMPACTED_LOB VARCHAR2(500) NULL,
    IMPACTED_SEAL VARCHAR2(500) NULL,
    ACCOUNTABLE_SEAL VARCHAR2(100) NULL,
    APPTEL VARCHAR2(50) NULL,
    RAISED_BY VARCHAR2(20) NULL,
    STATE VARCHAR2(20) NULL,
    CHANGE_CAUSED VARCHAR2(1) NULL,
    CHANGE_VENDOR VARCHAR2(1) NULL,
    CHANGE_NUMBER VARCHAR2(20) NULL,
    VENDOR_DETAILS VARCHAR2(500) NULL,
    CONSTRAINT unique_incident_team UNIQUE (INCIDENT_NUMBER, TEAM)
);

CREATE SEQUENCE incident_tracker_seq START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE TRIGGER incident_tracker_trigger
BEFORE INSERT ON incident_tracker
FOR EACH ROW
WHEN (NEW.ID IS NULL)
BEGIN
    SELECT incident_tracker_seq.NEXTVAL INTO :NEW.ID FROM DUAL;
END;
/


if oracle12g

@Id

private Long id;

@Id
@SequenceGenerator(name = "incident_seq", sequenceName = "incident_tracker_seq", allocationSize = 1)
@GeneratedValue(strategy = GenerationType.SEQUENCE, generator = "incident_seq")
private Long id;
